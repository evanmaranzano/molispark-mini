- [ ] **Step 2: 写 index.js（完整实现 view/like/unlike/comment）**

`cloudfunctions/interact/index.js`：
```js
const cloud = require('wx-server-sdk');
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });

const db = cloud.database();
const _ = db.command;

exports.main = async (event) => {
  const { OPENID } = cloud.getWXContext();
  const openid = OPENID || '';
  if (!openid) return { success: false, error: '未登录' };

  const { action, postId, content } = event;
  if (!postId) return { success: false, error: '缺少 postId' };

  try {
    if (action === 'view') {
      await db.collection('posts').doc(postId).update({ data: { views: _.inc(1) } });
      return { success: true };
    }

    if (action === 'like') {
      const existing = await db.collection('likes').where({ postId, openid }).limit(1).get();
      if (existing.data.length === 0) {
        await db.collection('likes').add({ data: { postId, openid, createdAt: db.serverDate() } });
        await db.collection('posts').doc(postId).update({ data: { likes: _.inc(1) } });
      }
      return { success: true, liked: true };
    }

    if (action === 'unlike') {
      const existing = await db.collection('likes').where({ postId, openid }).limit(1).get();
      if (existing.data.length > 0) {
        await db.collection('likes').doc(existing.data[0]._id).remove();
        await db.collection('posts').doc(postId).update({ data: { likes: _.inc(-1) } });
      }
      return { success: true, liked: false };
    }

    if (action === 'comment') {
      if (!content) return { success: false, error: '缺少 content' };
      const result = await db.collection('comments').add({
        data: { postId, openid, content, createdAt: db.serverDate() },
      });
      await db.collection('posts').doc(postId).update({ data: { commentCount: _.inc(1) } });
      return { success: true, commentId: result._id };
    }

    return { success: false, error: `未知 action: ${action}` };
  } catch (err) {
    console.error('interact failed:', err);
    return { success: false, error: err.message };
  }
};
```

async function callInteract(payload) {
  if (!isCloudReady()) return null;
  try {
    const res = await wx.cloud.callFunction({ name: 'interact', data: payload });
    return res.result;
  } catch (err) {
    console.error('callInteract failed:', err);

```js
async function toggleLike(postId, openid) {
  const existing = await query('likes', { postId, openid }, { limit: 1 });
  const willLike = existing.length === 0;

  if (isCloudReady()) {
    const result = await callInteract({ action: willLike ? 'like' : 'unlike', postId });
    if (result && result.success) {
      return { liked: result.liked };
    }
    // interact 失败则降级本地
  }

  if (willLike) {
    await add('likes', { postId, openid, createdAt: serverDate() });
    await updateById('posts', postId, { likes: _.inc(1) });
    return { liked: true };
  }
  await removeById('likes', existing[0]._id);
  await updateById('posts', postId, { likes: _.inc(-1) });
  return { liked: false };
}
```

**说明：** 需两个体验成员账号（A 作者、B 互动者），在微信开发者工具预览或体验版真机进行。

- [ ] **Step 1: A 登录 + 完善资料 + 发帖**

A 扫码进入 → 完善昵称头像 → 发布一篇帖子。预期：云数据库 `posts` 出现该记录，`users` 出现 A 的资料。

- [ ] **Step 2: B 登录 + 浏览 A 的帖子**

B 进入 → 打开 A 的帖子。预期：`posts.views` +1（查云数据库确认），detail 页浏览数 +1。

- [ ] **Step 3: B 点赞 + 取消点赞**

B 点赞 → 预期 `likes` 新增记录、`posts.likes` +1；再取消 → `likes` 记录删除、`posts.likes` -1。

- [ ] **Step 4: B 评论**

B 发评论 → 预期 `comments` 新增、`posts.commentCount` +1，detail 评论列表刷新显示。

- [ ] **Step 5: B 收藏**

B 收藏 → `collects` 新增。取消 → 删除。（不走 interact，客户端直接操作，权限允许。）

- [ ] **Step 6: 权限验证**

在控制台尝试用 B 的 openid 手动改 A 的帖子正文 → 应被权限拒绝（「仅创建者可写」）。

- [ ] **Step 7: 计数一致性核对**

云数据库核对：`posts.likes` == `likes` 集合中该 postId 的记录数；`posts.commentCount` == `comments` 集合中该 postId 的记录数。不一致则查 interact 云函数日志。
